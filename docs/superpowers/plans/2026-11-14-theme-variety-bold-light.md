> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship 17 theme presets (12 new gradient ones, 4 flat ones removed), mesh blooms on the chrome wash, and a saturated "bold frame" chrome with white text for every light theme.

**Architecture:** Everything is CSS custom properties in `src/app/globals.css`, selected by attributes on `<html>`. Preset blocks are generated by a committed script from OKLCH hue pairs and bound by `globals.contrast.test.ts`.

**Tech Stack:** Tailwind v4 (`@theme inline`), Next.js 16 App Router, Vitest, next-themes (`.dark` class on `<html>`).

```js
// Generates theme preset values from OKLCH hue pairs. Output is pasted into
// src/app/globals.css and src/lib/theme/presets.ts; globals.contrast.test.ts
// is the judge of every value, not this script.
const toLin = (v) => (v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4);
const toGam = (v) =>
  v <= 0.0031308 ? 12.92 * v : 1.055 * v ** (1 / 2.4) - 0.055;

/** OKLCH → linear sRGB (unclamped). */
function oklchToLinear(L, C, h) {
  const a = C * Math.cos((h * Math.PI) / 180);
  const b = C * Math.sin((h * Math.PI) / 180);
  const l = (L + 0.3963377774 * a + 0.2158037573 * b) ** 3;
  const m = (L - 0.1055613458 * a - 0.0638541728 * b) ** 3;
  const s = (L - 0.0894841775 * a - 1.291485548 * b) ** 3;
  return [
    4.0767416621 * l - 3.3077115913 * m + 0.2309699292 * s,
    -1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s,
    -0.0041960863 * l - 0.7034186147 * m + 1.707614701 * s,
  ];
}
const inGamut = (rgb) => rgb.every((v) => v >= -0.0005 && v <= 1.0005);

/** OKLCH → #rrggbb, lowering chroma until the colour is inside sRGB. */
export function oklchToHex(L, C, h) {
  let c = C;
  let rgb = oklchToLinear(L, c, h);
  while (!inGamut(rgb) && c > 0) {
    c -= 0.002;
    rgb = oklchToLinear(L, Math.max(c, 0), h);
  }
  return (
    "#" +
    rgb
      .map((v) =>
        Math.round(
          Math.min(1, Math.max(0, toGam(Math.min(1, Math.max(0, v))))) * 255,
        )
          .toString(16)
          .padStart(2, "0"),
      )
      .join("")
  );
}

const rgbOf = (hex) => [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16));
const lum = (hex) => {
  const [r, g, b] = rgbOf(hex).map((v) => toLin(v / 255));
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
};
export function contrastHex(a, b) {
  const [hi, lo] = [lum(a), lum(b)].sort((x, y) => y - x);
  return (hi + 0.05) / (lo + 0.05);
}
const overHex = (fg, alpha, bg) =>
  "#" +
  rgbOf(fg)
    .map((c, i) =>
      Math.round(alpha * c + (1 - alpha) * rgbOf(bg)[i])
        .toString(16)
        .padStart(2, "0"),
    )
    .join("");

/** Must match the values in globals.css (`:root:not(.dark) .chrome-scope`
 *  and the light --app-bloom). */
export const CHROME_MUTED_ALPHA = 0.86;
export const LIGHT_BLOOM = { tint: 0.3, peaks: [0.34, 0.28, 0.2] };

/** White muted chrome text on a stop under the two left-edge blooms. */
function mutedOnBloomed(stop, bloom1, bloom3) {
  const ground = overHex(
    bloom3,
    LIGHT_BLOOM.peaks[2],
    overHex(bloom1, LIGHT_BLOOM.peaks[0], stop),
  );
  return contrastHex(overHex("#ffffff", CHROME_MUTED_ALPHA, ground), ground);
}

/** Three wash stops, hue1 → hue2, each as light as it can be while white
 *  muted chrome text still clears 4.6:1 on it under bloom. */
export function solveBoldWash(
  h1,
  h2,
  cs = 1,
  bloom1 = "#ffffff",
  bloom3 = "#ffffff",
) {
  const targets = [
    [0.5, 0.16, h1],
    [0.43, 0.16, h1 + (((h2 - h1 + 540) % 360) - 180) * 0.6],
    [0.33, 0.12, h2],
  ];
  return targets.map(([L0, C, h]) => {
    let L = L0;
    let stop = oklchToHex(L, C * cs, h);
    while (mutedOnBloomed(stop, bloom1, bloom3) < 4.6 && L > 0.2) {
      L -= 0.005;
      stop = oklchToHex(L, C * cs, h);
    }
    return stop;
  });
}

if (process.argv[2] === "wash") {

    `--app-wash: linear-gradient(168deg, ${a} 0%, ${b} 46%, ${c} 100%);`,
  );
}
```
